```rust
use std::io::Write;
use std::process::{Command, Stdio};

const RAWK: &str = env!("CARGO_BIN_EXE_rawk");

/// Esegue rawk con `prog` come programma AWK e `stdin` come input grezzo.
/// Ritorna stdout grezzo (Vec<u8>) — il caller decide se confrontare in
/// bytes o convertire lossy per asserzioni testuali.
pub fn run_with_stdin(prog: &str, stdin: &[u8]) -> Vec<u8> {
    let mut child = Command::new(RAWK)
        .arg(prog)
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .spawn()
        .expect("spawn rawk");
    child
        .stdin
        .as_mut()
        .expect("stdin")
        .write_all(stdin)
        .expect("write stdin");
    let out = child.wait_with_output().expect("wait rawk");
    out.stdout
}
```

- [ ] **Step 1: Riscrivere `src/runner/fmt.rs` completo.**

Sostituire l'intero file con:

```rust
/*
 * Project: rawk (Rust AWK)
 * Authors: Francesco Tinti & Antigravity (Google Deepmind)
 * Description: Engine printf/sprintf di rawk. Phase 7.5 — byte-aware.
 *              `awk_sprintf(fmt: &[u8], args: &[AwkValue]) -> Vec<u8>`
 *              processa lo string format AWK byte-by-byte. Conversion
 *              numeriche (d/i/o/u/x/X/e/E/f/g/G) delegano a sprintf::sprintf!
 *              su ASCII puro e poi a Vec<u8>. %c emette il primo byte raw
 *              dell'argomento stringa. %s emette i byte raw senza UTF-8
 *              round-trip.
 */

use crate::types::AwkValue;

/// Sostituisce le sequenze `%…` in `fmt` con i valori convertiti di `args`.
/// Byte-aware: input e output sono `&[u8]` / `Vec<u8>`. Spec format string
/// non valido (conversion byte non in `diouxXeEfgGcs`) viene emesso letterale.
pub(super) fn awk_sprintf(fmt: &[u8], args: &[AwkValue]) -> Vec<u8> {
    let mut out: Vec<u8> = Vec::with_capacity(fmt.len());
    let mut i = 0;
    let mut arg_idx = 0;
    while i < fmt.len() {
        let b = fmt[i];
        if b != b'%' {
            out.push(b);
            i += 1;
            continue;
        }
        // Caso speciale %% senza arg
        if i + 1 < fmt.len() && fmt[i + 1] == b'%' {
            out.push(b'%');
            i += 2;
            continue;
        }
        // Accumula spec: flags + width + .precision + conversion
        let spec_start = i;
        i += 1;
        // Cerca un conversion byte. Tutti i byte intermedi (flags/digits/.)
        // li passiamo attraverso a sprintf::sprintf! senza validazione qui.
        let mut conv: Option<u8> = None;
        while i < fmt.len() {
            let c = fmt[i];
            i += 1;
            if b"diouxXeEfgGcs".contains(&c) {
                conv = Some(c);
                break;
            }
        }
        let spec_bytes = &fmt[spec_start..i];
        match conv {
            None => {
                // EOF dentro lo spec — emetti letterale e termina.
                out.extend_from_slice(spec_bytes);
                return out;
            }
            Some(c) => {
                let arg = args
                    .get(arg_idx)
                    .cloned()
                    .unwrap_or(AwkValue::Uninitialized);
                arg_idx += 1;
                format_one(spec_bytes, c, &arg, &mut out);
            }
        }
    }
    out
}

fn format_one(spec_bytes: &[u8], conv: u8, arg: &AwkValue, out: &mut Vec<u8>) {
    // Lo `spec` come `&str` è ASCII puro per costruzione (i byte ammessi nello
    // spec sono `%`, flags `-+ #0`, digit, `.`, e conversion byte: tutti ASCII).
    // Safe: passiamo a sprintf::sprintf! tramite `str::from_utf8_unchecked`
    // sarebbe possibile ma preferiamo `from_utf8` con expect, che è O(spec.len)
    // ma piccolo (raramente >10 byte).
    let spec = std::str::from_utf8(spec_bytes)
        .expect("awk_sprintf: format spec must be ASCII");
    match conv {
        b'd' | b'i' => {
            let s = sprintf::sprintf!(spec, arg.as_number() as i64).unwrap_or_default();
            out.extend_from_slice(s.as_bytes());
        }
        b'o' | b'x' | b'X' | b'u' => {
            let s = sprintf::sprintf!(spec, arg.as_number() as u64).unwrap_or_default();
            out.extend_from_slice(s.as_bytes());
        }
        b'e' | b'E' | b'f' | b'g' | b'G' => {
            let s = sprintf::sprintf!(spec, arg.as_number()).unwrap_or_default();
            out.extend_from_slice(s.as_bytes());
        }
        b'c' => {
            // Phase 7.5: %c emette il PRIMO BYTE raw dell'argomento stringa.
            // Se non-stringa (Number/Uninitialized), si converte il numero a u32
            // e si emette il byte basso (parità col comportamento legacy).
            let byte: u8 = match arg {
                AwkValue::String(s) | AwkValue::StrNum(s, _) if !s.is_empty() => s[0],
                _ => arg.as_number() as u32 as u8,
            };
            // Trattamento width/precision per %c: delega a sprintf::sprintf!
            // sostituendo `c` con `s` e passando un singolo byte come stringa
            // ASCII se possibile, altrimenti il byte viene push direttamente
            // (senza width formatting — lo spec %5c con byte alti è raro e non
            // coperto dai testcase XML).
            if byte < 0x80 && spec_bytes.len() == 2 {
                // Fast path: spec esattamente `%c`, byte ASCII → emetti raw.
                out.push(byte);
            } else if spec_bytes.len() == 2 {
                // Spec `%c`, byte alto → emetti raw.
                out.push(byte);
            } else {
                // Spec con width/flags (es. "%5c"). Per byte ASCII delega a
                // sprintf!("%5s", "X"); per byte alto, fallback raw push
                // (perdita width, ma byte preservato — caso edge, sopravvive
                // ai testcase XML che usano %c senza width).
                if byte < 0x80 {
                    let spec_s: String = spec.chars()
                        .map(|ch| if ch == 'c' { 's' } else { ch })
                        .collect();
                    let one = (byte as char).to_string();
                    let s = sprintf::sprintf!(&spec_s, one).unwrap_or_default();
                    out.extend_from_slice(s.as_bytes());
                } else {
                    out.push(byte);
                }
            }
        }
        b's' => {
            // Phase 7.5: %s emette `arg.as_string()` integro come bytes raw.
            let s_bytes = arg.as_string();
            if spec_bytes.len() == 2 {
                // Fast path: spec esattamente `%s` → emetti tutto raw.
                out.extend_from_slice(&s_bytes);
            } else {
                // Spec con width/precision. Applichiamo width/precision in modo
                // byte-aware:
                //   - precision `.N` tronca a N byte
                //   - width N pad a destra (default) o a sinistra (flag `-`)
                //     con spazi (0x20) o zeri (flag `0`).
                let (width, precision, left_align, zero_pad) = parse_s_flags(spec_bytes);
                let mut truncated: &[u8] = &s_bytes;
                if let Some(p) = precision {
                    if truncated.len() > p {
                        truncated = &truncated[..p];
                    }
                }
                let pad_count = width.saturating_sub(truncated.len());
                let pad_byte = if zero_pad { b'0' } else { b' ' };
                if left_align {
                    out.extend_from_slice(truncated);
                    for _ in 0..pad_count {
                        out.push(pad_byte);
                    }
                } else {
                    for _ in 0..pad_count {
                        out.push(pad_byte);
                    }
                    out.extend_from_slice(truncated);
                }
            }
        }
        _ => {
            // Conversion sconosciuto — già filtrato dal while sopra, non
            // dovrebbe accadere. Defensivo: emetti spec letterale.
            out.extend_from_slice(spec_bytes);
        }
    }
}

/// Parser dei flag dello spec `%s`: ritorna (width, precision, left_align, zero_pad).
/// `spec_bytes` include `%` e il conversion `s`. Esempi:
///   `%-5s`     -> (5, None, true, false)
///   `%.10s`    -> (0, Some(10), false, false)
///   `%05s`     -> (5, None, false, true)
///   `%-10.3s`  -> (10, Some(3), true, false)
fn parse_s_flags(spec_bytes: &[u8]) -> (usize, Option<usize>, bool, bool) {
    // Skip iniziale `%`.
    let mut i = 1;
    let mut left_align = false;
    let mut zero_pad = false;
    // Flags
    while i < spec_bytes.len() {
        match spec_bytes[i] {
            b'-' => {
                left_align = true;
                i += 1;
            }
            b'0' => {
                zero_pad = true;
                i += 1;
            }
            b'+' | b' ' | b'#' => {
                // Ignorati per %s (non semanticamente significativi).
                i += 1;
            }
            _ => break,
        }
    }
    // Width (sequenza di digit)
    let mut width: usize = 0;
    while i < spec_bytes.len() && spec_bytes[i].is_ascii_digit() {
        width = width * 10 + (spec_bytes[i] - b'0') as usize;
        i += 1;
    }
    // Precision: `.N`
    let mut precision: Option<usize> = None;
    if i < spec_bytes.len() && spec_bytes[i] == b'.' {
        i += 1;
        let mut p: usize = 0;
        while i < spec_bytes.len() && spec_bytes[i].is_ascii_digit() {
            p = p * 10 + (spec_bytes[i] - b'0') as usize;
            i += 1;
        }
        precision = Some(p);
    }
    (width, precision, left_align, zero_pad)
}
```

**Note di design**:
- Il path numerico (`%d/%i/%o/%u/%x/%X/%e/%E/%f/%g/%G`) delega a `sprintf::sprintf!` su `String` perché l'output è ASCII puro (i numeri non producono byte > 0x7F). Lo spec passa attraverso intatto a `sprintf::sprintf!`.
- Il path `%s` implementa width/precision **byte-aware**: la precisione tronca a N **byte**, la width padda a N **byte**. Questo è il comportamento POSIX byte-count desiderato (Phase 7 motivazione). Pre-7.5 il path delegava interamente a `sprintf::sprintf!(spec, lossy_string)`, che applicava width/precision **character-aware** su una stringa lossy: rotto su byte alti.
- Il path `%c` emette il **primo byte** dell'argomento stringa, non `chars().next()`. Per byte alti questo evita il collasso a U+FFFD. Per spec con width (raro nei testcase XML, ma valido), il fast-path emette solo il byte raw quando lo spec è esattamente `%c`; altrimenti delega a `sprintf::sprintf!` se ASCII, fallback raw push se byte alto.

```rust
/// Splitta `haystack` in segmenti separati da `sep` (substring match,
/// byte-exact). Se `sep` è vuoto, ritorna un singolo elemento contenente
/// l'intero `haystack` (parità col behavior `&str::split("")` AWK-side
/// non è ben definito, ma gawk con FS="" splitta per carattere; rawk
/// non supporta ancora FS="" — fuori scope 7.5).
fn split_bytes_by_separator<'a>(haystack: &'a [u8], sep: &[u8]) -> Vec<&'a [u8]> {
    if sep.is_empty() {
        return vec![haystack];
    }
    let mut out: Vec<&[u8]> = Vec::new();
    let mut start = 0;
    let mut i = 0;
    while i + sep.len() <= haystack.len() {
        if &haystack[i..i + sep.len()] == sep {
            out.push(&haystack[start..i]);
            i += sep.len();
            start = i;
        } else {
            i += 1;
        }
    }
    out.push(&haystack[start..]);
    out
}
```
